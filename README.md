import json
from textwrap import dedent

with open('profile.config.json', 'r', encoding='utf-8') as f:
    cfg = json.load(f)

name = cfg['name']
username = cfg['username']

# Helpers
phones = ' / '.join(cfg.get('phones', []))
portfolio = cfg.get('portfolio', '')
github = cfg.get('github', f'https://github.com/{username}')

skills = cfg['skills']

def bullet_list(items):
    return '\n'.join([f"• {i}" for i in items])

def section(title):
    return f"\n---\n\n## {title}\n"

# Build sections
about = dedent(f"""
# 👋 Hi, I'm **{name}**
**{cfg['title']}** · {cfg.get('location','')}

[🌐 Portfolio]({portfolio}) · [💻 GitHub]({github}) · ✉️ **{cfg['email']}** · 📞 **{phones}**

{section('🚀 About Me')}
{cfg['objective']}

{section('🧰 Tech Stack')}
**Languages & Frameworks:** {', '.join(skills['backend'])}  
**Frontend:** {', '.join(skills['frontend'])}  
**Data/Analytics:** {', '.join(skills['data'])}  
**Hardware/Robotics:** {', '.join(skills['hardware'])}  
**Tools:** {', '.join(skills['tools'])}  
**Soft Skills:** {', '.join(skills['soft'])}

{section('🧪 Projects')}
""")

projects_md = []
for p in cfg['projects']:
    line = f"**{p['name']}**  \n"
    if p.get('link'):
        line += f"🔗 GitHub: {p['link']}  \n"
    line += f"{p['desc']}\n"
    projects_md.append(line)

experience_md = []
for e in cfg['experience']:
    experience_md.append((
        f"**{e['role']} — {e['company']}**  \n"
        f"*{e['location']} | {e['dates']}*  \n"
        + '\n'.join([f"• {h}" for h in e['highlights']]) + '\n'
    ))

edu = cfg['education']
edu_md = dedent(f"""
**{edu['school']}**  \n{edu['degree']} | {edu['years']}
""")

stats = dedent(f"""

![GitHub Streak](https://streak-stats.demolab.com?user={username})
![GitHub Stats](https://github-readme-stats.vercel.app/api?username={username}&show_icons=true)
![Top Languages](https://github-readme-stats.vercel.app/api/top-langs/?username={username}&layout=compact)
""")

contact = dedent(f"""
- **Email:** {cfg['email']}  
- **Phone:** {phones}  
- **Portfolio:** {portfolio}  
- **GitHub:** {github}
""")
